## 1. 프로젝트 개요

### 1.1 프로젝트 목표

UST Project는 **모바일 매니퓰레이터 기반 VR 텔레오퍼레이션 및 모방 학습 시스템**을 구축하는 것을 목표로 합니다.

**핵심 기능:**
- VR (Meta Quest 3S) 기반 원격 조작
- 실시간 핸드 트래킹을 통한 로봇 제어
- 텔레오퍼레이션 데이터 수집 및 모방 학습
- ROS2 기반 실제 로봇 배포

### 1.2 하드웨어 구성

| 구성요소 | 사양 | 역할 |
|---------|-----|------|
| **모바일 베이스** | TurtleBot3 Waffle Pi | 2-DOF 차동 구동 |
| **매니퓰레이터** | OpenMANIPULATOR-X | 4-DOF 로봇 암 |
| **그리퍼** | OpenMANIPULATOR-X Gripper | 2-DOF 병렬 그리퍼 (mimic) |
| **3D 센서** | Intel RealSense D455 | RGB-D 카메라 |
| **3D LiDAR** | Livox MID-360 | 360도 LiDAR |
| **VR 장치** | Meta Quest 3S | 핸드 트래킹, 헤드 트래킹 |
| **GPU 서버** | RTX PRO 6000 (96GB) | 시뮬레이션, 학습, 추론 |

## 4. 핵심 구성요소 설계

### 4.1 로봇 구성 (ArticulationCfg)

#### 4.1.1 UST Mobile Manipulator Configuration

```python
# config/ust_mobile_manipulator_cfg.py

from isaaclab.assets import ArticulationCfg
from isaaclab.actuators import ImplicitActuatorCfg
import isaaclab.sim as sim_utils

UST_MOBILE_MANIPULATOR_CFG = ArticulationCfg(
    # USD 파일 경로
    spawn=sim_utils.UsdFileCfg(
        usd_path="./ust_ws/isaac_file/ust_project1.usd",
        rigid_props=sim_utils.RigidBodyPropertiesCfg(
            rigid_body_enabled=True,
            max_linear_velocity=1000.0,
            max_angular_velocity=1000.0,
            max_depenetration_velocity=100.0,
        ),
        articulation_props=sim_utils.ArticulationRootPropertiesCfg(
            enabled_self_collisions=False,
            solver_position_iteration_count=64,
            solver_velocity_iteration_count=4,
        ),
    ),
```

#### 4.3.2 Differential Drive Controller for Thumbstick

```python
# controllers/differential_drive_controller.py

import numpy as np
from dataclasses import dataclass
from typing import Tuple

@dataclass
class DifferentialDriveParams:
    """차동 구동 파라미터"""
    wheel_radius: float = 0.033  # TurtleBot3 바퀴 반지름
    wheel_base: float = 0.287    # 좌우 바퀴 간격
    max_linear_vel: float = 0.26
    max_angular_vel: float = 1.82


class DifferentialDriveController:
    """VR 썸스틱 → 차동 구동 바퀴 속도 변환"""

    def __init__(self, params: DifferentialDriveParams = None):
        self.params = params or DifferentialDriveParams()

    def thumbstick_to_wheel_velocities(
        self,
        thumbstick_x: float,
        thumbstick_y: float
    ) -> Tuple[float, float]:
        """
        썸스틱 입력을 바퀴 속도로 변환

        Args:
            thumbstick_x: 좌우 입력 (-1 ~ 1), 양수 = 우회전
            thumbstick_y: 전후 입력 (-1 ~ 1), 양수 = 전진

        Returns:
            (left_wheel_vel, right_wheel_vel): 바퀴 각속도 (rad/s)
        """
        # 선속도/각속도 계산
        linear_vel = thumbstick_y * self.params.max_linear_vel
        angular_vel = -thumbstick_x * self.params.max_angular_vel

        # 역기구학: v, w → 바퀴 속도
        v_left = (linear_vel - (self.params.wheel_base / 2) * angular_vel) / self.params.wheel_radius
        v_right = (linear_vel + (self.params.wheel_base / 2) * angular_vel) / self.params.wheel_radius

        return v_left, v_right

    def clamp_velocities(
        self,
        left_vel: float,
        right_vel: float,
        max_wheel_vel: float = 30.0
    ) -> Tuple[float, float]:
        """바퀴 속도 제한"""
        left_vel = np.clip(left_vel, -max_wheel_vel, max_wheel_vel)
        right_vel = np.clip(right_vel, -max_wheel_vel, max_wheel_vel)
        return left_vel, right_vel
```

#### 4.4.1 HDF5 Dataset Recorder

```python
# utils/hdf5_recorder.py

import h5py
import numpy as np
from datetime import datetime
from pathlib import Path
from typing import Dict, List, Any, Optional

class HDF5DatasetRecorder:
    """텔레오퍼레이션 데이터 HDF5 레코더"""

    def __init__(
        self,
        file_path: str,
        observation_keys: List[str],
        action_dim: int,
        include_images: bool = True
    ):
        self.file_path = Path(file_path)
        self.file_path.parent.mkdir(parents=True, exist_ok=True)

        self.observation_keys = observation_keys
        self.action_dim = action_dim
        self.include_images = include_images

        self.episodes: List[Dict[str, List]] = []
        self.current_episode: Optional[Dict[str, List]] = None

    def start_episode(self):
        """새 에피소드 시작"""
        self.current_episode = {
            "obs": [],
            "actions": [],
            "dones": [],
            "timestamps": [],
        }
        if self.include_images:
            self.current_episode["images_rgb"] = []
            self.current_episode["images_depth"] = []

    def add_step(
        self,
        obs: np.ndarray,
        action: np.ndarray,
        done: bool,
        rgb_image: Optional[np.ndarray] = None,
        depth_image: Optional[np.ndarray] = None,
        timestamp: Optional[float] = None
    ):
        """스텝 데이터 추가"""
        if self.current_episode is None:
            raise RuntimeError("Episode not started. Call start_episode() first.")

        self.current_episode["obs"].append(obs.copy())
        self.current_episode["actions"].append(action.copy())
        self.current_episode["dones"].append(done)
        self.current_episode["timestamps"].append(timestamp or datetime.now().timestamp())

        if self.include_images:
            if rgb_image is not None:
                self.current_episode["images_rgb"].append(rgb_image.copy())
            if depth_image is not None:
                self.current_episode["images_depth"].append(depth_image.copy())

    def end_episode(self, success: bool = True):
        """에피소드 종료"""
        if self.current_episode and len(self.current_episode["obs"]) > 0:
            self.current_episode["success"] = success
            self.episodes.append(self.current_episode)
        self.current_episode = None

    def save(self):
        """HDF5 파일로 저장"""
        with h5py.File(self.file_path, 'w') as f:
            # 메타데이터
            f.attrs["num_episodes"] = len(self.episodes)
            f.attrs["creation_time"] = datetime.now().isoformat()
            f.attrs["observation_keys"] = self.observation_keys
            f.attrs["action_dim"] = self.action_dim

            # 데이터 그룹
            data_grp = f.create_group("data")

            for i, ep in enumerate(self.episodes):
                ep_grp = data_grp.create_group(f"demo_{i}")

                # 관측/액션
                ep_grp.create_dataset("obs", data=np.array(ep["obs"]), compression="gzip")
                ep_grp.create_dataset("actions", data=np.array(ep["actions"]), compression="gzip")
                ep_grp.create_dataset("dones", data=np.array(ep["dones"]))
                ep_grp.create_dataset("timestamps", data=np.array(ep["timestamps"]))
                ep_grp.attrs["success"] = ep.get("success", True)
                ep_grp.attrs["length"] = len(ep["obs"])

                # 이미지 (있는 경우)
                if self.include_images and "images_rgb" in ep and len(ep["images_rgb"]) > 0:
                    ep_grp.create_dataset(
                        "images_rgb",
                        data=np.array(ep["images_rgb"]),
                        compression="gzip"
                    )
                if self.include_images and "images_depth" in ep and len(ep["images_depth"]) > 0:
                    ep_grp.create_dataset(
                        "images_depth",
                        data=np.array(ep["images_depth"]),
                        compression="gzip"
                    )

        print(f"Dataset saved: {self.file_path}")
        print(f"Total episodes: {len(self.episodes)}")

    def get_statistics(self) -> Dict[str, Any]:
        """데이터셋 통계"""
        if not self.episodes:
            return {}

        lengths = [len(ep["obs"]) for ep in self.episodes]
        successes = [ep.get("success", True) for ep in self.episodes]

        return {
            "num_episodes": len(self.episodes),
            "total_steps": sum(lengths),
            "avg_episode_length": np.mean(lengths),
            "success_rate": np.mean(successes),
            "min_length": min(lengths),
            "max_length": max(lengths),
        }
```

```python
from controllers.differential_drive_controller import DifferentialDriveController


def main():
    # 환경 설정 및 생성
    env_cfg = USTMobileManipulatorTeleopEnvCfg()
    env_cfg.scene.num_envs = args.num_envs
    env = ManagerBasedEnv(cfg=env_cfg)

    # 텔레오퍼레이션 장치 생성
    if args.teleop_device == "handtracking":
        teleop_cfg = USTTeleopDeviceCfg(enable_visualization=True)
        teleop_device = create_ust_teleop_device(teleop_cfg, sim_device=env.device)
    elif args.teleop_device == "keyboard":
        from isaaclab.devices import Se3Keyboard
        teleop_device = Se3Keyboard(pos_sensitivity=0.1, rot_sensitivity=0.1)
    else:
        from isaaclab.devices import Se3SpaceMouse
        teleop_device = Se3SpaceMouse(pos_sensitivity=0.1)

    # 차동 구동 컨트롤러
    base_controller = DifferentialDriveController()

    # 메인 루프
    print("\n=== UST Mobile Manipulator Teleoperation ===")
    print("Controls:")
    print("  Right Hand: Arm end-effector control")
    print("  Right Pinch: Gripper open/close")
    print("  Left Thumbstick: Mobile base control")
    print("  Palm Down Gesture: Reset environment")
    print("==========================================\n")

    obs, info = env.reset()

    while simulation_app.is_running():
        with torch.inference_mode():
            # 텔레오퍼레이션 명령 획득
            teleop_commands = teleop_device.advance()

            if teleop_commands is None:
                # 명령 없으면 제로 액션
                action = torch.zeros(env.num_envs, env.action_manager.total_action_dim, device=env.device)
            else:
                # 액션 구성
                action = torch.zeros(env.num_envs, env.action_manager.total_action_dim, device=env.device)

                # 암 SE3 명령 (위치 3D + 회전 3D)
                action[:, 0:6] = torch.tensor(teleop_commands["arm_delta_pose"], device=env.device)

                # 그리퍼 명령
                action[:, 6] = torch.tensor(teleop_commands["gripper_command"], device=env.device)

                # 베이스 속도 (썸스틱에서 변환)
                thumbstick = teleop_commands.get("thumbstick", (0.0, 0.0))
                left_vel, right_vel = base_controller.thumbstick_to_wheel_velocities(
                    thumbstick[0], thumbstick[1]
                )
                action[:, 7] = left_vel
                action[:, 8] = right_vel

            # 환경 스텝
            obs, info = env.step(action)

            # 리셋 처리
            if hasattr(teleop_device, 'reset_triggered') and teleop_device.reset_triggered:
                obs, info = env.reset()
                print("Environment reset.")

    env.close()


if __name__ == "__main__":
    main()
    simulation_app.close()
```

### 5.3 Phase 2: 데이터 수집

#### 5.3.1 데이터 수집 스크립트

```python
# scripts/record_demos.py

import argparse
import torch
from pathlib import Path
from datetime import datetime

from isaaclab.app import AppLauncher

parser = argparse.ArgumentParser(description="Record Teleoperation Demos")
parser.add_argument("--num_demos", type=int, default=20)
parser.add_argument("--dataset_path", type=str, default="./datasets")
parser.add_argument("--include_images", action="store_true")
AppLauncher.add_app_launcher_args(parser)
args = parser.parse_args()

app_launcher = AppLauncher(args)
simulation_app = app_launcher.app

from isaaclab.envs import ManagerBasedEnv
from config.ust_teleop_env_cfg import USTMobileManipulatorTeleopEnvCfg
from config.ust_teleop_device_cfg import USTTeleopDeviceCfg, create_ust_teleop_device
from controllers.differential_drive_controller import DifferentialDriveController
from utils.hdf5_recorder import HDF5DatasetRecorder


def main():
    # 환경 설정
    env_cfg = USTMobileManipulatorTeleopEnvCfg()
    env = ManagerBasedEnv(cfg=env_cfg)

    # 텔레오퍼레이션 장치
    teleop_cfg = USTTeleopDeviceCfg()
    teleop_device = create_ust_teleop_device(teleop_cfg, sim_device=env.device)
    base_controller = DifferentialDriveController()

    # 데이터 레코더
    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    dataset_file = Path(args.dataset_path) / f"ust_demos_{timestamp}.hdf5"

    recorder = HDF5DatasetRecorder(
        file_path=str(dataset_file),
        observation_keys=["joint_pos", "joint_vel", "ee_pos", "ee_quat", "gripper_pos"],
        action_dim=9,  # 6 (arm) + 1 (gripper) + 2 (base)
        include_images=args.include_images
    )

    # 수집 루프
    completed_demos = 0
    obs, info = env.reset()
    recorder.start_episode()

    print(f"\n=== Recording {args.num_demos} Demos ===")
    print("Perform tasks with VR controller.")
    print("Palm-down gesture to end episode.\n")

    while simulation_app.is_running() and completed_demos < args.num_demos:
        with torch.inference_mode():
            teleop_commands = teleop_device.advance()

            if teleop_commands is None:
                action = torch.zeros(env.num_envs, env.action_manager.total_action_dim, device=env.device)
                env.step(action)
                continue

            # 액션 구성
            action = torch.zeros(env.num_envs, 9, device=env.device)
            action[:, 0:6] = torch.tensor(teleop_commands["arm_delta_pose"], device=env.device)
            action[:, 6] = torch.tensor(teleop_commands["gripper_command"], device=env.device)

            thumbstick = teleop_commands.get("thumbstick", (0.0, 0.0))
            left_vel, right_vel = base_controller.thumbstick_to_wheel_velocities(thumbstick[0], thumbstick[1])
            action[:, 7] = left_vel
            action[:, 8] = right_vel

            # 데이터 기록
            obs_np = obs["policy"].cpu().numpy()[0]
            action_np = action.cpu().numpy()[0]

            recorder.add_step(
                obs=obs_np,
                action=action_np,
                done=False,
                rgb_image=obs.get("images_rgb", [None])[0] if args.include_images else None,
                depth_image=obs.get("images_depth", [None])[0] if args.include_images else None
            )

            # 환경 스텝
            obs, info = env.step(action)

            # 에피소드 종료 체크
            if hasattr(teleop_device, 'reset_triggered') and teleop_device.reset_triggered:
                recorder.end_episode(success=True)
                completed_demos += 1
                print(f"Demo {completed_demos}/{args.num_demos} completed.")

                if completed_demos < args.num_demos:
                    obs, info = env.reset()
                    recorder.start_episode()

    # 저장
    recorder.save()
    stats = recorder.get_statistics()
    print(f"\n=== Recording Complete ===")
    print(f"Total episodes: {stats['num_episodes']}")
    print(f"Total steps: {stats['total_steps']}")
    print(f"Average length: {stats['avg_episode_length']:.1f}")
    print(f"Success rate: {stats['success_rate']:.1%}")

    env.close()


if __name__ == "__main__":
    main()
    simulation_app.close()
```

```python
# scripts/run_ros2_bridge.py

"""ROS2 Bridge를 통한 실제 로봇 연동"""

import rclpy
from rclpy.node import Node
from sensor_msgs.msg import JointState
from geometry_msgs.msg import Twist
from std_msgs.msg import Float64MultiArray


class USTRobotBridge(Node):
    """Isaac Sim ↔ 실제 로봇 ROS2 브릿지"""

    def __init__(self):
        super().__init__('ust_robot_bridge')

        # 퍼블리셔
        self.cmd_vel_pub = self.create_publisher(Twist, '/cmd_vel', 10)
        self.arm_cmd_pub = self.create_publisher(Float64MultiArray, '/arm_controller/command', 10)
        self.gripper_cmd_pub = self.create_publisher(Float64MultiArray, '/gripper_controller/command', 10)

        # 서브스크라이버
        self.joint_state_sub = self.create_subscription(
            JointState, '/joint_states', self.joint_state_callback, 10
        )

        self.get_logger().info('UST Robot Bridge initialized')

    def joint_state_callback(self, msg: JointState):
        """실제 로봇 관절 상태 수신"""
        # Isaac Sim으로 상태 전달 로직
        pass

    def send_base_command(self, linear_vel: float, angular_vel: float):
        """모바일 베이스 명령 전송"""
        msg = Twist()
        msg.linear.x = linear_vel
        msg.angular.z = angular_vel
        self.cmd_vel_pub.publish(msg)

    def send_arm_command(self, joint_positions: list):
        """암 관절 명령 전송"""
        msg = Float64MultiArray()
        msg.data = joint_positions
        self.arm_cmd_pub.publish(msg)

    def send_gripper_command(self, gripper_position: float):
        """그리퍼 명령 전송"""
        msg = Float64MultiArray()
        msg.data = [gripper_position]
        self.gripper_cmd_pub.publish(msg)


def main(args=None):
    rclpy.init(args=args)
    bridge = USTRobotBridge()
    rclpy.spin(bridge)
    bridge.destroy_node()
    rclpy.shutdown()


if __name__ == '__main__':
    main()
```
