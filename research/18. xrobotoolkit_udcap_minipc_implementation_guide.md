## 3. XRoboToolkit C++ SDK → Python 브릿지

### 3.1 아키텍처

XRoboToolkit PC Service는 **C++ SDK만 제공**하며 Python Pybind는 없다. 따라서 Python에서 데이터에 접근하려면 브릿지를 구현해야 한다.

```
XRoboToolkit PC Service (C++ 프로세스)
        │
        │ gRPC / 내부 프로토콜
        ▼
C++ SDK (PXREAInit → 콜백)
        │
        │ 방법 선택
        ├── (A) C++ 데모 프로그램 → JSON stdout → Python subprocess
        ├── (B) C++ → TCP/ZeroMQ 소켓 → Python 수신
        └── (C) C++ → 공유 메모리 / 파일 → Python 읽기
```

### 3.2 방법 A: C++ 데모 → JSON stdout 파이프 (가장 간단)

```bash
# 1. C++ ConsoleDemo 빌드
cd ~/XRoboToolkit-PC-Service/SDK/Demo/CppSrc/ConsoleDemo

# CMakeLists.txt 확인 및 빌드
mkdir build && cd build
cmake .. -DCMAKE_PREFIX_PATH="$HOME/XRoboToolkit-PC-Service/SDK"
make -j$(nproc)

# 2. 데모 실행하여 JSON 출력 확인
./ConsoleDemo
# → PICO 연결 시 JSON 형태의 트래킹 데이터 출력
```

```python
# xrt_bridge.py - Python 측 브릿지
"""XRoboToolkit C++ Demo → Python JSON 브릿지"""

import subprocess
import json
import threading
import queue

class XRoboToolkitBridge:
    """C++ ConsoleDemo를 subprocess로 실행하고 JSON 데이터를 파싱하는 브릿지"""

    def __init__(self, demo_path: str):
        self.demo_path = demo_path
        self.data_queue = queue.Queue(maxsize=10)
        self._running = False

    def start(self):
        self._running = True
        self._thread = threading.Thread(target=self._reader_loop, daemon=True)
        self._thread.start()

    def _reader_loop(self):
        proc = subprocess.Popen(
            [self.demo_path],
            stdout=subprocess.PIPE,
            stderr=subprocess.DEVNULL,
            text=True,
            bufsize=1  # line buffered
        )
        for line in proc.stdout:
            if not self._running:
                proc.kill()
                break
            line = line.strip()
            if line.startswith('{'):
                try:
                    data = json.loads(line)
                    if self.data_queue.full():
                        self.data_queue.get_nowait()  # 오래된 데이터 버림
                    self.data_queue.put_nowait(data)
                except json.JSONDecodeError:
                    pass

    def get_latest(self) -> dict | None:
        """최신 트래킹 데이터 반환 (없으면 None)"""
        data = None
        while not self.data_queue.empty():
            data = self.data_queue.get_nowait()
        return data

    def stop(self):
        self._running = False
```

### 3.3 방법 B: C++ → TCP 소켓 → Python (더 안정적)

```cpp
// xrt_tcp_bridge.cpp - C++ 소켓 서버 (SDK 콜백 → TCP JSON)
#include "PXREASDK.h"
#include <sys/socket.h>
#include <netinet/in.h>
#include <cstring>
#include <string>

int server_fd;
int client_fd = -1;

// XRoboToolkit SDK 콜백
void onDeviceState(const char* jsonStr, int length) {
    if (client_fd > 0) {
        std::string data(jsonStr, length);
        data += "\n";  // 줄바꿈 구분자
        send(client_fd, data.c_str(), data.size(), MSG_NOSIGNAL);
    }
}

int main() {
    // TCP 서버 소켓 (포트 7777)
    server_fd = socket(AF_INET, SOCK_STREAM, 0);
    struct sockaddr_in addr;
    addr.sin_family = AF_INET;
    addr.sin_addr.s_addr = INADDR_ANY;
    addr.sin_port = htons(7777);
    bind(server_fd, (struct sockaddr*)&addr, sizeof(addr));
    listen(server_fd, 1);

    // XRoboToolkit SDK 초기화
    PXREAInit("127.0.0.1", onDeviceState);

    // 클라이언트 연결 대기
    while (true) {
        printf("Waiting for Python client on port 7777...\n");
        client_fd = accept(server_fd, NULL, NULL);
        printf("Python client connected\n");
        // 연결 유지 (끊어지면 재연결 대기)
    }

    PXREADeinit();
    return 0;
}
```

```python
# xrt_tcp_receiver.py - Python TCP 수신기
"""XRoboToolkit TCP 브릿지 수신기 (Docker 내에서도 실행 가능)"""

import socket
import json
import threading

class XRTTCPReceiver:
    """XRoboToolkit C++ TCP 브릿지에서 JSON 데이터를 수신"""

    def __init__(self, host: str = "localhost", port: int = 7777):
        self.host = host
        self.port = port
        self.latest_data = None
        self._lock = threading.Lock()
        self._running = False

    def start(self):
        self._running = True
        self._thread = threading.Thread(target=self._recv_loop, daemon=True)
        self._thread.start()

    def _recv_loop(self):
        while self._running:
            try:
                sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
                sock.connect((self.host, self.port))
                buffer = ""
                while self._running:
                    chunk = sock.recv(65536).decode('utf-8')
                    if not chunk:
                        break
                    buffer += chunk
                    while '\n' in buffer:
                        line, buffer = buffer.split('\n', 1)
                        if line.strip():
                            try:
                                data = json.loads(line)
                                with self._lock:
                                    self.latest_data = data
                            except json.JSONDecodeError:
                                pass
            except (ConnectionRefusedError, ConnectionResetError):
                import time
                time.sleep(1)  # 재연결 대기

    def get_latest(self) -> dict | None:
        with self._lock:
            return self.latest_data

    def stop(self):
        self._running = False
```

- CPU: Intel N95/N100 이상 (UDCAP 소프트웨어는 경량)
- RAM: 8GB
- 저장: 128GB+ (Windows + UDCAP + SteamVR)
- USB: USB 3.0 x1 이상 (UDCAP 동글용)
- WiFi: 5GHz 지원 (동일 LAN 필수)
- Windows 11 프리인스톨

### 4.2 미니PC 초기 셋업

```
1. 미니PC 전원 ON → Windows 11 초기 설정

3. 고정 IP 설정 (권장):
   - 설정 → 네트워크 → WiFi → IP 설정 → 수동

   - 서브넷: 255.255.255.0
   - 게이트웨이: 192.168.0.1
4. 방화벽에서 UDP 포트 허용:
   - Windows Defender 방화벽 → 고급 설정
   - 인바운드 규칙 → 새 규칙
   - UDP, 포트: 39539 (VMC), 9000 (OSC)
   - 허용
```

## 5. Ubuntu 호스트: 데이터 수신 인프라

### 5.1 VMC 수신기 (Python)

```python
# vmc_receiver.py
"""VMC (Virtual Motion Capture) 프로토콜 수신기
UDCAP Software → UDP → 이 스크립트 → Isaac Lab
"""

from pythonosc import dispatcher, osc_server
import threading
import json

class VMCReceiver:
    """VMC/OSC 프로토콜로 UDCAP 핑거 데이터 수신

    VMC 프로토콜은 OSC의 확장으로, /VMC/Ext/Bone/Pos 등의 주소를 사용.
    UDCAP은 각 손가락 관절의 회전값을 이 프로토콜로 전송.
    """

    def __init__(self, listen_ip: str = "0.0.0.0", listen_port: int = 39539):
        self.listen_ip = listen_ip
        self.listen_port = listen_port
        self.left_hand_joints = {}   # joint_name → (qx, qy, qz, qw)
        self.right_hand_joints = {}
        self._lock = threading.Lock()

    def start(self):
        disp = dispatcher.Dispatcher()
        # VMC 본(bone) 포즈 수신
        disp.map("/VMC/Ext/Bone/Pos", self._on_bone_pos)
        # UDCAP 커스텀 주소 (확인 필요)
        disp.map("/udcap/*", self._on_udcap_data)
        # 모든 메시지 로깅 (디버그용)
        disp.set_default_handler(self._on_any)

        self._server = osc_server.ThreadingOSCUDPServer(
            (self.listen_ip, self.listen_port), disp
        )
        self._thread = threading.Thread(target=self._server.serve_forever, daemon=True)
        self._thread.start()
        print(f"VMC Receiver listening on {self.listen_ip}:{self.listen_port}")

    def _on_bone_pos(self, address, *args):
        """VMC 본 포즈 콜백: /VMC/Ext/Bone/Pos name px py pz qx qy qz qw"""
        if len(args) >= 8:
            bone_name = args[0]
            px, py, pz = args[1], args[2], args[3]
            qx, qy, qz, qw = args[4], args[5], args[6], args[7]

            with self._lock:
                # 왼손/오른손 판별 (bone_name 기준)
                if "Left" in bone_name:
                    self.left_hand_joints[bone_name] = (qx, qy, qz, qw)
                elif "Right" in bone_name:
                    self.right_hand_joints[bone_name] = (qx, qy, qz, qw)

    def _on_udcap_data(self, address, *args):
        """UDCAP 커스텀 OSC 주소 콜백"""
        print(f"[UDCAP] {address}: {args}")
        # 실제 UDCAP의 OSC 주소 패턴은 테스트로 확인 필요

    def _on_any(self, address, *args):
        """디버그: 모든 OSC 메시지 출력"""
        print(f"[VMC/OSC] {address}: {args[:5]}...")  # 처음 5개만

    def get_finger_data(self) -> dict:
        """최신 핑거 데이터 반환"""
        with self._lock:
            return {
                "left": dict(self.left_hand_joints),
                "right": dict(self.right_hand_joints),
            }

    def stop(self):
        self._server.shutdown()


# 테스트 실행
if __name__ == "__main__":
    receiver = VMCReceiver(listen_port=39539)
    receiver.start()

    import time
    while True:
        time.sleep(1)
        data = receiver.get_finger_data()
        print(f"Left joints: {len(data['left'])}, Right joints: {len(data['right'])}")
```

### 5.2 통합 데이터 수집기

```python
# unified_teleop_bridge.py
"""통합 텔레오퍼레이션 데이터 브릿지

XRoboToolkit (Body/Hand) + UDCAP (Finger) → TCP → Docker (Isaac Lab)
호스트에서 실행, Docker로 통합 데이터 전달
"""

import json
import socket
import threading
import time

class UnifiedTeleopBridge:
    """XRoboToolkit + UDCAP 데이터를 통합하여 Isaac Lab에 전달"""

    def __init__(self,
                 xrt_host: str = "localhost",
                 xrt_port: int = 7777,
                 vmc_port: int = 39539,
                 output_port: int = 8888):
        """
        Args:
            xrt_host: XRoboToolkit C++ 브릿지 호스트
            xrt_port: XRoboToolkit C++ 브릿지 포트
            vmc_port: UDCAP VMC 수신 포트
            output_port: Isaac Lab으로 전달할 TCP 포트
        """
        self.xrt_receiver = None    # XRoboToolkit 데이터
        self.vmc_receiver = None    # UDCAP 데이터
        self.output_port = output_port
        self._running = False

    def start(self):
        self._running = True
        # ... 각 수신기 시작 + 통합 루프

    def _merge_and_send(self):
        """90Hz로 통합 데이터를 TCP로 전송"""
        server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        server.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
        server.bind(("0.0.0.0", self.output_port))
        server.listen(1)

        while self._running:
            print(f"[Bridge] Waiting for Isaac Lab on port {self.output_port}...")
            client, addr = server.accept()
            print(f"[Bridge] Isaac Lab connected from {addr}")

            try:
                while self._running:
                    # XRoboToolkit 데이터 (body + hand + controller)
                    xrt_data = self.xrt_receiver.get_latest() if self.xrt_receiver else {}
                    # UDCAP 핑거 데이터
                    finger_data = self.vmc_receiver.get_finger_data() if self.vmc_receiver else {}

                    merged = {
                        "timestamp": time.time_ns(),
                        "xrt": xrt_data,          # head, body, hands, controllers, trackers
                        "udcap_fingers": finger_data,  # left/right hand joint rotations
                    }

                    msg = json.dumps(merged) + "\n"
                    client.sendall(msg.encode('utf-8'))
                    time.sleep(1.0 / 90)  # 90Hz

            except (BrokenPipeError, ConnectionResetError):
                print("[Bridge] Isaac Lab disconnected, waiting for reconnect...")
```

## 6. Docker 컨테이너: Isaac Lab 통합

### 6.1 통합 텔레오퍼레이션 디바이스

```python
# ust_config/ust_fullbody_teleop_device.py (신규 파일, Docker 내 Isaac Lab)
"""PICO 4 Ultra Full-Body 텔레오퍼레이션 디바이스

데이터 소스:
- XRoboToolkit: Head 6DoF, Body 24pt, Hands 26j×2, Controllers
- UDCAP: Finger joint rotations (VMC/OSC 경유)
- 통합 브릿지: TCP port 8888 (호스트에서 실행)
"""

import socket
import json
import threading
import torch

class PICOFullBodyTeleopDevice:
    """PICO 4 Ultra + Motion Tracker + UDCAP 통합 텔레오퍼레이션 디바이스"""

    def __init__(self, bridge_host: str = "localhost", bridge_port: int = 8888):
        self.bridge_host = bridge_host
        self.bridge_port = bridge_port
        self._latest_data = None
        self._lock = threading.Lock()
        self._connected = False

    def start(self):
        """브릿지에 연결하고 데이터 수신 시작"""
        self._thread = threading.Thread(target=self._recv_loop, daemon=True)
        self._thread.start()

    def _recv_loop(self):
        while True:
            try:
                sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
                sock.connect((self.bridge_host, self.bridge_port))
                self._connected = True
                buffer = ""

                while True:
                    chunk = sock.recv(65536).decode('utf-8')
                    if not chunk:
                        break
                    buffer += chunk
                    while '\n' in buffer:
                        line, buffer = buffer.split('\n', 1)
                        if line.strip():
                            data = json.loads(line)
                            with self._lock:
                                self._latest_data = data
            except Exception:
                self._connected = False
                import time
                time.sleep(1)

    def advance(self) -> dict:
        """최신 텔레오퍼레이션 데이터 반환

        Returns:
            {
                "head_pose": (pos3, quat4),
                "body_joints": [(pos3, quat4), ...],  # 24 포인트
                "left_hand_pose": (pos3, quat4),       # 컨트롤러 위치
                "right_hand_pose": (pos3, quat4),
                "left_hand_joints": [(qx,qy,qz,qw),...], # 26 관절 (XRT)
                "right_hand_joints": [...],
                "left_finger_angles": {...},  # UDCAP (있으면)
                "right_finger_angles": {...},
                "left_trigger": float,
                "right_trigger": float,
                "left_grip": float,
                "right_grip": float,
            }
        """
        with self._lock:
            if self._latest_data is None:
                return None
            return self._parse_data(self._latest_data)

    def _parse_data(self, raw: dict) -> dict:
        """통합 브릿지 JSON → 텔레오퍼레이션 명령 변환"""
        xrt = raw.get("xrt", {})
        fingers = raw.get("udcap_fingers", {})

        result = {
            "head_pose": self._extract_pose(xrt.get("head", {}).get("pose")),
            "body_joints": [
                self._extract_pose(j.get("pose"))
                for j in xrt.get("body", {}).get("joints", [])
            ],
        }

        # 컨트롤러 데이터
        ctrl = xrt.get("controllers", {})
        for side in ("left", "right"):
            c = ctrl.get(side, {})
            result[f"{side}_hand_pose"] = self._extract_pose(c.get("pose"))
            result[f"{side}_trigger"] = c.get("trigger", 0.0)
            result[f"{side}_grip"] = c.get("grip", 0.0)

        # XRoboToolkit 핸드 트래킹 (26 관절)
        hands = xrt.get("hands", {})
        for side in ("left", "right"):
            h = hands.get(side, {})
            result[f"{side}_hand_joints"] = [
                self._extract_rotation(j.get("pose"))
                for j in h.get("joints", [])
            ]

        # UDCAP 핑거 (있으면 오버라이드)
        if fingers.get("left"):
            result["left_finger_angles"] = fingers["left"]
        if fingers.get("right"):
            result["right_finger_angles"] = fingers["right"]

        return result

    @staticmethod
    def _extract_pose(pose: dict | None):
        if not pose:
            return None
        pos = pose.get("position", {})
        rot = pose.get("rotation", {})
        return (
            (pos.get("x", 0), pos.get("y", 0), pos.get("z", 0)),
            (rot.get("x", 0), rot.get("y", 0), rot.get("z", 0), rot.get("w", 1)),
        )

    @staticmethod
    def _extract_rotation(pose: dict | None):
        if not pose:
            return (0, 0, 0, 1)
        rot = pose.get("rotation", {})
        return (rot.get("x", 0), rot.get("y", 0), rot.get("z", 0), rot.get("w", 1))
```
