### 3.4 OpenXRDevice 및 Retargeter 상세 설정

```python
# config/xr_teleop_cfg.py
from isaaclab.devices import OpenXRDevice, OpenXRDeviceCfg, DeviceBase
from isaaclab.devices.openxr import XrCfg
from isaaclab.devices.openxr.retargeters import (
    Se3AbsRetargeter, Se3AbsRetargeterCfg,
    Se3RelRetargeter, Se3RelRetargeterCfg,
    GripperRetargeter, GripperRetargeterCfg
)

def create_single_arm_teleop_device():
    """단일 암 텔레오퍼레이션 디바이스 생성"""

    # XR 앵커 설정 (로봇 베이스 위치)
    xr_cfg = XrCfg(
        anchor_pos=[0.0, 0.0, 0.0],
        anchor_rot=[1.0, 0.0, 0.0, 0.0]  # wxyz 쿼터니언
    )

    # 절대 위치 리타게터 (손 위치 → EE 위치)
    position_retargeter = Se3AbsRetargeter(
        Se3AbsRetargeterCfg(
            bound_hand=DeviceBase.TrackingTarget.HAND_RIGHT,
            zero_out_xy_rotation=True,    # Z축 회전만 허용 (안정성)
            use_wrist_position=False,     # 핀치 포지션 사용 (정밀도↑)
            use_wrist_rotation=True,
            enable_visualization=True,
            device="cuda"
        )
    )

    # 그리퍼 리타게터 (핀치 제스처)
    gripper_retargeter = GripperRetargeter(
        GripperRetargeterCfg(
            bound_hand=DeviceBase.TrackingTarget.HAND_RIGHT
        )
    )

    # OpenXR 디바이스 생성
    device = OpenXRDevice(
        OpenXRDeviceCfg(xr_cfg=xr_cfg),
        retargeters=[position_retargeter, gripper_retargeter]
    )

    return device


def create_dual_arm_teleop_device():
    """듀얼 암 텔레오퍼레이션 디바이스 생성"""

    xr_cfg = XrCfg(
        anchor_pos=[0.0, 0.0, 0.0],
        anchor_rot=[1.0, 0.0, 0.0, 0.0]
    )

    # 왼팔 리타게터
    left_position = Se3AbsRetargeter(Se3AbsRetargeterCfg(
        bound_hand=DeviceBase.TrackingTarget.HAND_LEFT,
        use_wrist_rotation=True,
        use_wrist_position=True,
    ))
    left_gripper = GripperRetargeter(GripperRetargeterCfg(
        bound_hand=DeviceBase.TrackingTarget.HAND_LEFT,
    ))

    # 오른팔 리타게터
    right_position = Se3AbsRetargeter(Se3AbsRetargeterCfg(
        bound_hand=DeviceBase.TrackingTarget.HAND_RIGHT,
        use_wrist_rotation=True,
        use_wrist_position=True,
    ))
    right_gripper = GripperRetargeter(GripperRetargeterCfg(
        bound_hand=DeviceBase.TrackingTarget.HAND_RIGHT,
    ))

    device = OpenXRDevice(
        OpenXRDeviceCfg(xr_cfg=xr_cfg),
        retargeters=[
            left_position, left_gripper,
            right_position, right_gripper
        ]
    )

    return device
```

### 4.4 스키드 스티어 베이스 컨트롤러

```python
# controllers/skid_steer_controller.py
import numpy as np
from isaacsim.robot.wheeled_robots.controllers.differential_controller import DifferentialController
from isaacsim.core.utils.types import ArticulationAction

class SkidSteerController:
    """4륜 Skid-Steer 모바일 베이스 컨트롤러"""

    def __init__(self,
                 wheel_radius: float = 0.05,
                 wheel_base: float = 0.24,
                 max_linear_vel: float = 0.5,
                 max_angular_vel: float = 1.0):
        """
        Args:
            wheel_radius: 바퀴 반지름 (m)
            wheel_base: 좌우 바퀴 간격 (m)
            max_linear_vel: 최대 선속도 (m/s)
            max_angular_vel: 최대 각속도 (rad/s)
        """
        self.controller = DifferentialController(
            name="skid_steer",
            wheel_radius=wheel_radius,
            wheel_base=wheel_base
        )
        self.max_linear_vel = max_linear_vel
        self.max_angular_vel = max_angular_vel
        # 조인트 순서: FL, FR, RL, RR
        self.wheel_indices = [0, 1, 2, 3]

    def compute_wheel_velocities(self,
                                  linear_vel: float,
                                  angular_vel: float) -> np.ndarray:
        """선속도/각속도 → 바퀴 속도 변환"""
        # 속도 제한
        linear_vel = np.clip(linear_vel, -self.max_linear_vel, self.max_linear_vel)
        angular_vel = np.clip(angular_vel, -self.max_angular_vel, self.max_angular_vel)

        action = self.controller.forward([linear_vel, angular_vel])
        left_vel = action.joint_velocities[0]
        right_vel = action.joint_velocities[1]

        # Skid-steer: 좌측 바퀴 동일, 우측 바퀴 동일
        return np.array([left_vel, right_vel, left_vel, right_vel])

    def from_thumbstick(self, thumbstick_x: float, thumbstick_y: float) -> np.ndarray:
        """VR 컨트롤러 썸스틱 → 바퀴 속도"""
        linear_vel = thumbstick_y * self.max_linear_vel
        angular_vel = -thumbstick_x * self.max_angular_vel
        return self.compute_wheel_velocities(linear_vel, angular_vel)
```

```python
# scripts/record_teleop_demos.py
import argparse
import torch
import h5py
import numpy as np
from datetime import datetime

from isaaclab.app import AppLauncher

parser = argparse.ArgumentParser()
parser.add_argument("--teleop_device", type=str, default="handtracking")
parser.add_argument("--dataset_file", type=str, default="./datasets/demo.hdf5")
parser.add_argument("--num_demos", type=int, default=10)
AppLauncher.add_app_launcher_args(parser)
args = parser.parse_args()

app_launcher = AppLauncher(args)
simulation_app = app_launcher.app

from config.mobile_manipulator_env_cfg import MobileManipulatorTeleopEnvCfg
from config.xr_teleop_cfg import create_single_arm_teleop_device
from controllers.skid_steer_controller import SkidSteerController
from isaaclab.envs import ManagerBasedEnv

class HDF5DatasetRecorder:
    """HDF5 데이터셋 레코더"""

    def __init__(self, file_path: str):
        self.file_path = file_path
        self.episodes = []
        self.current_episode = None

    def start_episode(self):
        self.current_episode = {
            "obs": [],
            "actions": [],
            "dones": []
        }

    def add_step(self, obs: np.ndarray, action: np.ndarray, done: bool):
        self.current_episode["obs"].append(obs)
        self.current_episode["actions"].append(action)
        self.current_episode["dones"].append(done)

    def end_episode(self):
        if self.current_episode and len(self.current_episode["obs"]) > 0:
            self.episodes.append(self.current_episode)
        self.current_episode = None

    def save(self):
        with h5py.File(self.file_path, 'w') as f:
            data_grp = f.create_group("data")
            for i, ep in enumerate(self.episodes):
                ep_grp = data_grp.create_group(f"demo_{i}")
                ep_grp.create_dataset("obs", data=np.array(ep["obs"]))
                ep_grp.create_dataset("actions", data=np.array(ep["actions"]))
                ep_grp.create_dataset("dones", data=np.array(ep["dones"]))

            # 메타데이터
            f.attrs["num_demos"] = len(self.episodes)
            f.attrs["timestamp"] = datetime.now().isoformat()

def main():
    # 환경 생성
    env_cfg = MobileManipulatorTeleopEnvCfg()
    env = ManagerBasedEnv(cfg=env_cfg)

    # 텔레오퍼레이션 디바이스 초기화
    if args.teleop_device == "handtracking":
        device = create_single_arm_teleop_device()
    else:
        from isaaclab.devices import Se3SpaceMouse
        device = Se3SpaceMouse(pos_sensitivity=0.1)

    # 베이스 컨트롤러
    base_controller = SkidSteerController(
        wheel_radius=0.05,
        wheel_base=0.24
    )

    # 데이터 레코더
    recorder = HDF5DatasetRecorder(args.dataset_file)

    # 메인 루프
    completed_demos = 0
    obs, info = env.reset()
    recorder.start_episode()

    print(f"\n=== 데모 수집 시작 (목표: {args.num_demos}개) ===")
    print("핀치 제스처로 그리퍼 제어, 왼손 썸스틱으로 베이스 이동")
    print("에피소드 완료 후 리셋 제스처 수행\n")

    while simulation_app.is_running() and completed_demos < args.num_demos:
        with torch.inference_mode():
            # VR 명령 획득
            commands = device.advance()

            if commands is None:
                # 명령 없으면 제로 액션
                action = torch.zeros(env.num_envs, env.action_manager.total_action_dim)
                env.step(action)
                continue

            # 액션 구성: [arm_pose(6), gripper(1), base_vel(4)]
            action = torch.zeros(env.num_envs, 11, device=env.device)

            # 암 SE3 명령 (위치 3D + 회전 3D)
            action[:, :6] = torch.tensor(commands[:6])

            # 그리퍼 명령
            action[:, 6] = commands[6] if len(commands) > 6 else 0.0

            # 베이스 명령 (썸스틱에서)
            # TODO: VR 컨트롤러 썸스틱 데이터 획득
            wheel_vels = base_controller.compute_wheel_velocities(0.0, 0.0)
            action[:, 7:11] = torch.tensor(wheel_vels)

            # 데이터 기록
            recorder.add_step(
                obs["policy"].cpu().numpy()[0],
                action.cpu().numpy()[0],
                done=False
            )

            # 환경 스텝
            obs, info = env.step(action)

            # 리셋 콜백 처리
            if hasattr(device, 'reset_triggered') and device.reset_triggered:
                recorder.end_episode()
                completed_demos += 1
                print(f"데모 {completed_demos}/{args.num_demos} 완료")

                if completed_demos < args.num_demos:
                    obs, info = env.reset()
                    recorder.start_episode()

    # 저장
    recorder.save()
    print(f"\n데이터셋 저장 완료: {args.dataset_file}")
    print(f"총 {completed_demos}개 에피소드 수집됨")

    env.close()

if __name__ == "__main__":
    main()
    simulation_app.close()
```

```python
# scripts/convert_to_lerobot.py
"""Isaac Lab HDF5 → LeRobot 데이터셋 변환"""

import h5py
import json
from pathlib import Path
import numpy as np
from datasets import Dataset, Features, Sequence, Value
import pyarrow as pa
import pyarrow.parquet as pq

def convert_isaaclab_to_lerobot(
    input_hdf5: str,
    output_dir: str,
    dataset_name: str = "isaac_mobile_manip"
):
    """Isaac Lab HDF5를 LeRobot 형식으로 변환"""

    output_path = Path(output_dir) / dataset_name
    output_path.mkdir(parents=True, exist_ok=True)

    with h5py.File(input_hdf5, 'r') as f:
        data_grp = f["data"]

        all_episodes = []
        for demo_name in data_grp.keys():
            demo = data_grp[demo_name]
            obs = demo["obs"][:]
            actions = demo["actions"][:]

            for i in range(len(obs)):
                all_episodes.append({
                    "observation.state": obs[i].tolist(),
                    "action": actions[i].tolist(),
                    "episode_index": int(demo_name.split("_")[1]),
                    "frame_index": i,
                    "timestamp": i * 0.02,  # 50Hz 가정
                })

    # Parquet로 저장
    table = pa.Table.from_pylist(all_episodes)
    pq.write_table(table, output_path / "data.parquet")

    # 메타데이터
    metadata = {
        "fps": 50,
        "num_episodes": len(set(ep["episode_index"] for ep in all_episodes)),
        "features": {
            "observation.state": {"shape": [obs.shape[-1]]},
            "action": {"shape": [actions.shape[-1]]},
        }
    }

    with open(output_path / "meta.json", 'w') as f:
        json.dump(metadata, f, indent=2)

    print(f"LeRobot 데이터셋 저장: {output_path}")

if __name__ == "__main__":
    convert_isaaclab_to_lerobot(
        "./datasets/mimic_1k.hdf5",
        "~/.cache/huggingface/lerobot/",
        "isaac_mobile_manip"
    )
```

### 8.1 CloudXR 연결 문제

| 증상 | 원인 | 해결 방법 |
|-----|------|---------|
| 검은 화면 | GPU 인덱스 불일치 | `export NV_GPU_INDEX=0` |
| XR_ERROR_INSTANCE_LOST | CloudXR 런타임 먼저 종료 | 런타임 재시작 |
| 48010 연결 실패 | 포트 차단 | 방화벽 규칙 확인 |
| 높은 지연 (>100ms) | 무선 네트워크 | 5GHz WiFi, 유선 서버 |
| 핸드 트래킹 불안정 | 조명 부족 | 조명 개선, Quest 3S IR 센서 활용 |

### 8.2 시뮬레이션 문제

| 증상 | 원인 | 해결 방법 |
|-----|------|---------|
| 물리 불안정 | 타임스텝 너무 큼 | `sim.dt = 0.005` 이하로 |
| 로봇 떨림 | 액추에이터 게인 부적절 | stiffness/damping 조정 |
| IK 실패 | 도달 불가 영역 | 작업 공간 범위 확인 |
| 느린 시뮬레이션 | GPU 메모리 부족 | num_envs 감소 |

### 8.3 데이터 수집 문제

| 증상 | 원인 | 해결 방법 |
|-----|------|---------|
| 빈 HDF5 파일 | 에피소드 종료 미감지 | 리셋 트리거 확인 |
| 데이터 동기화 문제 | obs/action 타이밍 불일치 | 동일 타임스텝에서 수집 |
| 대용량 파일 | 이미지 데이터 포함 | 압축 또는 샘플링 |

|-----|------|---------|
