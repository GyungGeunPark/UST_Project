## 1. 프로젝트 개요

### 1.1 태스크 정의

**목표**: Unitree G1 29DOF 휴머노이드 로봇이 가정집 환경에서 이동 중 갑자기 넘어지는 노인을 감지하고 안전하게 부축하는 모방학습 파이프라인 구축

**시나리오 상세**:
```
┌─────────────────────────────────────────────────────────────────┐
│                    노인 부축 태스크 시나리오                        │
│                                                                 │
│  Phase A: 관찰 및 감지                                            │
│  ┌──────────┐                                                   │
│  │ 로봇이   │  → 노인이 거실/주방에서 이동 중                        │
│  │ 대기/관찰│  → 갑작스러운 균형 상실 감지                           │
│  └──────────┘  → 낙상 예측/인식                                   │
│                                                                 │
│  Phase B: 접근 및 개입                                            │
│  ┌──────────┐                                                   │
│  │ 빠른 접근│  → 로코모션으로 노인 근처 이동                         │

│  └──────────┘  → 안전 접촉 지점 계산                               │
```

```python
from isaaclab.scene import InteractiveSceneCfg
from isaaclab.assets import AssetBaseCfg, RigidObjectCfg
from isaaclab.sim import UsdFileCfg, GroundPlaneCfg
import isaaclab.sim as sim_utils

@configclass
class ElderlyHomeSceneCfg(InteractiveSceneCfg):
    """노인 가정집 시뮬레이션 환경"""

    # ==================== 환경 크기 설정 ====================
    env_spacing = 6.0  # 환경 간 간격 (충분히 넓게)

    # ==================== 지면 ====================
    ground = AssetBaseCfg(
        prim_path="/World/GroundPlane",
        spawn=GroundPlaneCfg(
            size=(20.0, 20.0),
            physics_material=sim_utils.RigidBodyMaterialCfg(
                static_friction=0.8,
                dynamic_friction=0.6,
            ),
        ),
    )

    # ==================== 거실 가구 ====================
    # 소파
    sofa = AssetBaseCfg(
        prim_path="{ENV_REGEX_NS}/Furniture/Sofa",
        spawn=UsdFileCfg(
            usd_path="path/to/interior_agent/sofa_001.usd",
            scale=(1.0, 1.0, 1.0),
        ),
        init_state=AssetBaseCfg.InitialStateCfg(
            pos=(2.0, 0.0, 0.0),
            rot=(1.0, 0.0, 0.0, 0.0),
        ),
    )

    # 커피 테이블
    coffee_table = AssetBaseCfg(
        prim_path="{ENV_REGEX_NS}/Furniture/CoffeeTable",
        spawn=UsdFileCfg(
            usd_path="path/to/interior_agent/coffee_table_001.usd",
        ),
        init_state=AssetBaseCfg.InitialStateCfg(
            pos=(1.0, 0.0, 0.0),
        ),
    )

    # 의자 (노인이 앉을 수 있는)
    chair = RigidObjectCfg(
        prim_path="{ENV_REGEX_NS}/Furniture/Chair",
        spawn=UsdFileCfg(
            usd_path="path/to/interior_agent/armchair_001.usd",
            rigid_props=sim_utils.RigidBodyPropertiesCfg(
                disable_gravity=False,
                kinematic_enabled=True,  # 고정된 가구
            ),
        ),
        init_state=RigidObjectCfg.InitialStateCfg(
            pos=(-1.5, 1.5, 0.0),
            rot=(0.707, 0.0, 0.0, 0.707),  # 90도 회전
        ),
    )

    # ==================== 주방 영역 ====================
    kitchen_counter = AssetBaseCfg(
        prim_path="{ENV_REGEX_NS}/Kitchen/Counter",
        spawn=UsdFileCfg(
            usd_path="path/to/interior_agent/kitchen_counter_001.usd",
        ),
        init_state=AssetBaseCfg.InitialStateCfg(
            pos=(-3.0, -2.0, 0.0),
        ),
    )

    # ==================== 장애물 (낙상 위험 요소) ====================
    # 러그 (걸려 넘어질 수 있는)
    rug = AssetBaseCfg(
        prim_path="{ENV_REGEX_NS}/Hazards/Rug",
        spawn=UsdFileCfg(
            usd_path="path/to/interior_agent/area_rug_001.usd",
        ),
        init_state=AssetBaseCfg.InitialStateCfg(
            pos=(0.0, 0.0, 0.001),  # 약간 들림
        ),
    )

    # 전선/코드 (장애물)
    cable = AssetBaseCfg(
        prim_path="{ENV_REGEX_NS}/Hazards/Cable",
        spawn=UsdFileCfg(
            usd_path="path/to/cable_hazard.usd",
        ),
        init_state=AssetBaseCfg.InitialStateCfg(
            pos=(0.5, -0.5, 0.0),
        ),
    )

    # ==================== 조명 ====================

            intensity=1500.0,

        ),
        init_state=AssetBaseCfg.InitialStateCfg(
```

```python
from isaaclab.managers import RandomizationTermCfg

@configclass
class ElderlyHomeRandomizationCfg:
    """환경 도메인 랜덤화 설정"""

    # 가구 위치 랜덤화

        params={

            "rotation_range": (0.0, 0.0, 0.2),  # roll, pitch, yaw 범위
        },
        mode="reset",
    )

    # 조명 조건 랜덤화
    lighting = RandomizationTermCfg(
        func=randomize_lighting,
        params={
            "intensity_range": (800.0, 2500.0),
            "color_range": ((0.9, 0.85, 0.8), (1.0, 1.0, 0.95)),
        },
        mode="reset",
    )

    # 바닥 마찰 계수 랜덤화
    floor_friction = RandomizationTermCfg(
        func=randomize_floor_friction,
        params={
            "static_friction_range": (0.4, 1.0),
            "dynamic_friction_range": (0.3, 0.8),
        },
        mode="reset",
    )

    # 장애물 위치 랜덤화
    hazard_position = RandomizationTermCfg(

            "num_hazards": (1, 3),

        },
        mode="reset",
```

```python
# elderly_character_cfg.py

from omni.anim.people import PeopleSimulation

@configclass

    # 체형 파라미터 (SMPL 호환)
```

#### 4.2.2 합성 낙상 모션 생성

```python
# synthetic_fall_generator.py

import numpy as np
from scipy.interpolate import CubicSpline

class SyntheticFallGenerator:
    """합성 낙상 모션 생성기"""

    def __init__(self, fps: int = 30):
        self.fps = fps
        self.fall_duration = {
            FallType.FORWARD_TRIP: (0.8, 1.5),      # 0.8~1.5초
            FallType.FORWARD_SLIP: (0.5, 1.0),
            FallType.BACKWARD_SLIP: (0.6, 1.2),
            FallType.SLOW_COLLAPSE: (2.0, 4.0),
            FallType.STUMBLE_TO_FALL: (1.5, 3.0),
        }

    def generate_fall_trajectory(
        self,
        fall_type: FallType,
        initial_pose: np.ndarray,
        final_pose: np.ndarray,
        duration: float = None,
    ) -> np.ndarray:
        """
        낙상 궤적 생성

        Args:
            fall_type: 낙상 유형
            initial_pose: 초기 자세 (SMPL pose parameters)
            final_pose: 최종 자세 (바닥에 누운 상태)
            duration: 낙상 시간 (초)

        Returns:
            trajectory: (T, 72) 형태의 관절 궤적
        """
        if duration is None:
            min_dur, max_dur = self.fall_duration[fall_type]
            duration = np.random.uniform(min_dur, max_dur)

        num_frames = int(duration * self.fps)

        # 낙상 역학 모델링
        trajectory = self._apply_fall_dynamics(
            initial_pose, final_pose, num_frames, fall_type
        )

        # 자연스러운 모션을 위한 노이즈 추가
        trajectory = self._add_motion_noise(trajectory, fall_type)

        return trajectory

    def _apply_fall_dynamics(
        self,
        initial: np.ndarray,
        final: np.ndarray,
        num_frames: int,
        fall_type: FallType,
    ) -> np.ndarray:
        """물리 기반 낙상 역학 적용"""

        t = np.linspace(0, 1, num_frames)

        if fall_type in [FallType.FORWARD_TRIP, FallType.FORWARD_SLIP]:
            # 가속 낙하: 중력 영향
            # s(t) = 0.5 * g * t^2 정규화
            progress = t ** 2
        elif fall_type == FallType.SLOW_COLLAPSE:
            # 감속 하강: 의식적 제어
            progress = 1 - (1 - t) ** 2
        else:
            # 기본: 시그모이드 전환
            progress = 1 / (1 + np.exp(-10 * (t - 0.5)))

        # 보간
        trajectory = np.outer(1 - progress, initial) + np.outer(progress, final)

        return trajectory

    def _add_motion_noise(
        self,
        trajectory: np.ndarray,
        fall_type: FallType,
    ) -> np.ndarray:
        """자연스러운 모션 노이즈 추가"""

        noise_scale = {
            FallType.FORWARD_TRIP: 0.05,
            FallType.STUMBLE_TO_FALL: 0.08,
            FallType.SLOW_COLLAPSE: 0.02,
        }.get(fall_type, 0.03)

        noise = np.random.normal(0, noise_scale, trajectory.shape)

        # 시간 축 스무딩
        from scipy.ndimage import gaussian_filter1d
        noise = gaussian_filter1d(noise, sigma=2, axis=0)

        return trajectory + noise
```

```python
class ElderlyState(Enum):
```

### 6.2 보상 함수 상세 구현

```python
# reward_functions.py

import torch
from isaaclab.managers import RewardTermCfg

def reward_approach_target(
    env,
    robot_body: str,
    target: str,
    optimal_distance: float,
) -> torch.Tensor:
    """
    목표(노인)에 접근하는 보상

    최적 거리에 가까울수록 높은 보상
    """
    robot_pos = env.robot.data.body_pos[:, env.robot.find_body_idx(robot_body)]
    target_pos = env.elderly.get_position()

    distance = torch.norm(robot_pos[:, :2] - target_pos[:, :2], dim=1)

    # 가우시안 형태의 보상
    reward = torch.exp(-((distance - optimal_distance) ** 2) / (2 * 0.3 ** 2))

    return reward


def reward_safe_contact_force(
    env,
    min_force: float,
    max_force: float,
    optimal_force: float,
) -> torch.Tensor:
    """
    안전한 접촉력 범위 유지 보상

    최적 힘에 가까울수록 높은 보상
    너무 약하거나 강하면 패널티
    """
    contact_forces = env.force_sensor.data.force_matrix

    # 양손의 총 접촉력
    left_force = torch.norm(contact_forces[:, 0, :], dim=1)
    right_force = torch.norm(contact_forces[:, 1, :], dim=1)
    total_force = left_force + right_force

    reward = torch.zeros_like(total_force)

    # 최적 범위 내
    in_range = (total_force >= min_force) & (total_force <= max_force)
    reward[in_range] = 1.0 - torch.abs(total_force[in_range] - optimal_force) / optimal_force

    # 범위 외 페널티
    too_low = total_force < min_force
    too_high = total_force > max_force

    reward[too_low] = -0.5
    reward[too_high] = -(total_force[too_high] - max_force) / max_force

    return reward


def reward_support_success(
    env,
    elderly_stabilized_threshold: float,
) -> torch.Tensor:
    """
    부축 성공 보상

    노인의 각속도가 임계값 이하로 안정화되면 보상
    """
    elderly_angular_vel = env.elderly.get_angular_velocity()
    angular_speed = torch.norm(elderly_angular_vel, dim=1)

    # 낙상 중이었다가 안정화된 경우만 보상
    was_falling = env.elderly_state_buffer[:, -1] == ElderlyState.FALLING.value
    is_stable = angular_speed < elderly_stabilized_threshold

    success = was_falling & is_stable

    return success.float() * 10.0  # 큰 보상


def penalty_excessive_force(
    env,
    force_threshold: float,
) -> torch.Tensor:
    """
    과도한 접촉력 패널티

    인간에게 위험할 수 있는 높은 힘 적용 시 페널티
    """
    contact_forces = env.force_sensor.data.force_matrix

    left_force = torch.norm(contact_forces[:, 0, :], dim=1)
    right_force = torch.norm(contact_forces[:, 1, :], dim=1)
    max_force = torch.maximum(left_force, right_force)

    excessive = max_force > force_threshold

    penalty = torch.zeros_like(max_force)
    penalty[excessive] = (max_force[excessive] - force_threshold) / force_threshold

    return -penalty


def reward_bilateral_support(
    env,
    force_balance_threshold: float,
) -> torch.Tensor:
    """
    양팔 균형 사용 보상

    좌우 팔의 힘이 균형있게 사용될 때 보상
    """
    contact_forces = env.force_sensor.data.force_matrix

    left_force = torch.norm(contact_forces[:, 0, :], dim=1)
    right_force = torch.norm(contact_forces[:, 1, :], dim=1)

    total_force = left_force + right_force + 1e-6  # 분모 0 방지
    balance_ratio = torch.abs(left_force - right_force) / total_force

    # 균형 잡힐수록 높은 보상
    reward = 1.0 - balance_ratio / force_balance_threshold
    reward = torch.clamp(reward, 0.0, 1.0)

    # 양팔 모두 사용 중일 때만 보상
    both_in_contact = (left_force > 1.0) & (right_force > 1.0)

    return reward * both_in_contact.float()
```

### 9.1 임피던스 제어 구현

```python
# impedance_controller.py

import torch
from isaaclab.controllers import DifferentialIKController

class ComplianceImpedanceController:
    """
    부축 태스크를 위한 임피던스 제어기

    참조:
    - "Compliant Force Control for Robots: A Survey" (MDPI, 2025)
    - MIT E-BAR 연구
    """

    def __init__(
        self,
        stiffness: torch.Tensor,   # 강성 행렬 (6,)
        damping: torch.Tensor,     # 감쇠 행렬 (6,)
        inertia: torch.Tensor,     # 관성 행렬 (6,)
    ):
        """
        Args:
            stiffness: 카테시안 공간 강성 [Kx, Ky, Kz, Krx, Kry, Krz]
            damping: 카테시안 공간 감쇠
            inertia: 원하는 관성 (옵션)
        """
        self.K = torch.diag(stiffness)  # (6, 6)
        self.D = torch.diag(damping)    # (6, 6)
        self.M = torch.diag(inertia)    # (6, 6)

        # 상태 버퍼
        self.x_d = None       # 목표 위치
        self.x_d_dot = None   # 목표 속도
        self.x = None         # 현재 위치
        self.x_dot = None     # 현재 속도
        self.f_ext = None     # 외력

    def compute_command(
        self,
        desired_pose: torch.Tensor,       # (N, 7) pos + quat
        current_pose: torch.Tensor,       # (N, 7)
        current_velocity: torch.Tensor,   # (N, 6)
        external_force: torch.Tensor,     # (N, 6)
    ) -> torch.Tensor:
        """
        임피던스 제어 명령 계산

        임피던스 관계: M * x_ddot + D * x_dot + K * x = f_ext

        Returns:
            wrench: (N, 6) 카테시안 힘/토크 명령
        """
        batch_size = desired_pose.shape[0]

        # 위치 오차
        pos_error = desired_pose[:, :3] - current_pose[:, :3]

        # 방향 오차 (쿼터니언 → 축-각)
        quat_error = self._quaternion_error(
            desired_pose[:, 3:7],
            current_pose[:, 3:7]
        )

        # 전체 포즈 오차
        pose_error = torch.cat([pos_error, quat_error], dim=1)  # (N, 6)

        # 속도 오차 (목표 속도 = 0 가정)
        vel_error = -current_velocity

        # 임피던스 제어 법칙
        # f = K * (x_d - x) + D * (x_d_dot - x_dot) + f_ext_compensation
        wrench = (
            torch.matmul(pose_error.unsqueeze(1), self.K.T).squeeze(1) +
            torch.matmul(vel_error.unsqueeze(1), self.D.T).squeeze(1)
        )

        # 외력 보상 (옵션)
        if self.compensate_external_force:
            wrench = wrench - external_force

        return wrench

    def _quaternion_error(
        self,
        quat_desired: torch.Tensor,
        quat_current: torch.Tensor,
    ) -> torch.Tensor:
        """쿼터니언 오차를 축-각 오차로 변환"""
        # q_error = q_desired * q_current^(-1)
        quat_current_inv = self._quaternion_inverse(quat_current)
        quat_error = self._quaternion_multiply(quat_desired, quat_current_inv)

        # 축-각 추출
        angle = 2 * torch.acos(torch.clamp(quat_error[:, 0], -1, 1))
        axis = quat_error[:, 1:4] / (torch.sin(angle / 2).unsqueeze(1) + 1e-8)

        return axis * angle.unsqueeze(1)


class VariableImpedanceController(ComplianceImpedanceController):
    """
    가변 임피던스 제어기

    상황에 따라 강성/감쇠 동적 조정
    """

    def __init__(self, ...):
        super().__init__(...)

        # 강성 범위
        self.K_min = torch.tensor([50, 50, 50, 10, 10, 10])
        self.K_max = torch.tensor([500, 500, 500, 100, 100, 100])

        # 감쇠 범위
        self.D_min = torch.tensor([5, 5, 5, 1, 1, 1])
        self.D_max = torch.tensor([50, 50, 50, 10, 10, 10])

    def adapt_impedance(
        self,
        task_phase: str,
        contact_force: torch.Tensor,
        human_stability: torch.Tensor,
    ):
        """
        태스크 단계 및 상황에 따른 임피던스 조정

        Args:
            task_phase: "approach", "contact", "support", "stabilize"
            contact_force: 현재 접촉력
            human_stability: 노인 안정성 지표 (0-1)
        """
        if task_phase == "approach":
            # 접근 시: 높은 강성 (정밀 위치 제어)
            alpha = 0.8
        elif task_phase == "contact":
            # 초기 접촉: 낮은 강성 (부드러운 접촉)
            alpha = 0.2
        elif task_phase == "support":
            # 부축 중: 중간 강성 (힘 제어)
            alpha = 0.4 + 0.3 * human_stability
        elif task_phase == "stabilize":
            # 안정화: 강성 점진적 증가
            alpha = 0.5 + 0.4 * human_stability
        else:
            alpha = 0.5

        # 강성/감쇠 보간
        self.K = torch.diag(self.K_min + alpha * (self.K_max - self.K_min))
        self.D = torch.diag(self.D_min + alpha * (self.D_max - self.D_min))
```

### 9.2 안전 제약 학습

```python
# safe_rl_wrapper.py

import torch
from isaaclab.envs import DirectRLEnv

class SafeRLWrapper:
    """
    안전 제약 강화학습 래퍼

    참조: "Safety Control for Service Robots" (ScienceDirect, 2025)
    """

    def __init__(
        self,
        env: DirectRLEnv,
        max_contact_force: float = 100.0,   # N
        max_joint_torque: float = 50.0,     # Nm
        max_joint_velocity: float = 3.0,    # rad/s
        max_tilt_angle: float = 0.4,        # rad
    ):
        self.env = env
        self.constraints = {
            "max_contact_force": max_contact_force,
            "max_joint_torque": max_joint_torque,
            "max_joint_velocity": max_joint_velocity,
            "max_tilt_angle": max_tilt_angle,
        }

        # 위반 카운터
        self.violation_counts = {k: 0 for k in self.constraints}

    def step(self, actions: torch.Tensor):
        """안전 제약 적용된 스텝"""

        # 1. 행동 클리핑 (사전 제약)
        safe_actions = self._clip_actions(actions)

        # 2. 환경 스텝
        obs, rewards, dones, infos = self.env.step(safe_actions)

        # 3. 제약 위반 검사 (사후 검사)
        violations = self._check_violations()

        # 4. 제약 위반 페널티
        constraint_penalty = self._compute_constraint_penalty(violations)
        rewards = rewards - constraint_penalty

        # 5. 심각한 위반 시 조기 종료
        severe_violations = self._check_severe_violations(violations)
        dones = dones | severe_violations

        # 정보 업데이트
        infos["constraint_violations"] = violations

        return obs, rewards, dones, infos

    def _clip_actions(self, actions: torch.Tensor) -> torch.Tensor:
        """행동을 안전 범위 내로 클리핑"""
        # 관절 속도 제한 반영
        max_delta = self.constraints["max_joint_velocity"] * self.env.step_dt
        clipped = torch.clamp(
            actions,
            self.env.last_actions - max_delta,
            self.env.last_actions + max_delta,
        )
        return clipped

    def _check_violations(self) -> dict:
        """제약 위반 검사"""
        violations = {}

        # 접촉력 검사
        contact_forces = self.env.force_sensor.data.force_matrix
        max_force = contact_forces.abs().max(dim=-1).values.max(dim=-1).values
        violations["contact_force"] = max_force > self.constraints["max_contact_force"]

        # 관절 토크 검사
        joint_torques = self.env.robot.data.joint_torques
        max_torque = joint_torques.abs().max(dim=-1).values
        violations["joint_torque"] = max_torque > self.constraints["max_joint_torque"]

        # 관절 속도 검사
        joint_velocities = self.env.robot.data.joint_velocities
        max_velocity = joint_velocities.abs().max(dim=-1).values
        violations["joint_velocity"] = max_velocity > self.constraints["max_joint_velocity"]

        # 기울기 검사
        root_quat = self.env.robot.data.root_quat
        tilt_angle = self._compute_tilt_angle(root_quat)
        violations["tilt_angle"] = tilt_angle > self.constraints["max_tilt_angle"]

        return violations

    def _compute_constraint_penalty(self, violations: dict) -> torch.Tensor:
        """제약 위반에 따른 페널티 계산"""
        penalty = torch.zeros(self.env.num_envs, device=self.env.device)

        penalty_weights = {
            "contact_force": 1.0,
            "joint_torque": 0.5,
            "joint_velocity": 0.3,
            "tilt_angle": 0.8,
        }

        for name, violated in violations.items():
            penalty += penalty_weights[name] * violated.float()

        return penalty
```
